# cat=site; type=int; label=Site Root Page ID:Root page ID of the TYPO3 site this MCP server is responsible for. Determines which domains are allowed and which base URL is used for CLI/OAuth. Set to 0 to allow all configured sites (no domain validation).
siteRootPageId = 0

# cat=mcp/tables; type=string; label=Additional read-only tables: Comma-separated list of non-workspace-capable tables to expose as read-only via MCP tools. These tables become available in ReadTable but cannot be written. Example: tx_blog_domain_model_author,tx_myext_domain_model_setting
additionalReadOnlyTables = sys_file

# cat=mcp/tables; type=string; label=Additional standalone tables: Comma-separated list of tables marked as hideTable=true in TCA that should still be exposed as independent tables instead of being embedded into their parent's inline relation. # cat=logging; type=boolean; label=Verbose request logging: Write a per-request trace to the PHP error log (method, headers, query parameters, the resolved status code). Credentials are redacted either way. Off by default because it is several lines per request and buries everything else — switch it on to diagnose one installation, then switch it back off. Authentication failures are logged regardless of this setting.
debugLogging = 0

# cat=mcp/oauth; type=int+; label=Access token lifetime (seconds): Upper bound on how long an OAuth session survives. A token cannot outlive this even when it is never used again, so shortening it is the lever against sessions that stay valid long after anybody last touched them. Clients run through the authorization flow again afterwards. Default 2592000 (30 days); 28800 is 8 hours.
tokenLifetime = 2592000

# cat=mcp/oauth; type=int+; label=Authorization code lifetime (seconds): How long an issued authorization code can be exchanged for an access token. Covers only the seconds between consent and the client's token request, so it is short by design. Default 600 (10 minutes).
authorizationCodeLifetime = 600
